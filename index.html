function saveGoal() { const name = document.getElementById('zielName').value.trim(), target = parseFloat(document.getElementById('zielBetrag').value) || 0; if(!name || target <= 0) return alert('Name und Zielbetrag eingeben!'); const id = 'goal_' + Date.now(), type = document.getElementById('zielTypeSelect').value; goals.push({ id, name, target, type, fixedMonthly: parseFloat(document.getElementById('zielFixedMonthly').value)||0, startMonth: getSelectedMonthKey(), endDate: document.getElementById('zielAutoDivideCheckbox').checked?document.getElementById('zielEndMonthInput').value:null, manualCurrent: parseFloat(document.getElementById('zielStart').value)||0, manualOffset: parseFloat(document.getElementById('zielStart').value)||0, capAtTarget: document.getElementById('zielCapAtTargetCheckbox').checked }); if(document.getElementById('zielAutoUrlaubCheckbox').checked) { urlaubeList.push({ goalId: id, startDate: document.getElementById('zielStartDateInput').value, endDate: document.getElementById('zielEndDateInput').value }); activeUrlaubGoalId = id; } saveLocally(); document.getElementById('zielName').value = ''; document.getElementById('zielBetrag').value = ''; renderGoals(); renderHistory(); checkUrlaubTabVisibility(); updateSubKontoDropdowns(); alert('Ziel / Unterkonto angelegt! 🎯'); }
